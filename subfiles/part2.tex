\begin{frame}[t,mycolor=digiPH_leaf,mytitle=standard,dark]
	Pondasi adalah elemen struktur bawah \textit{(sub-structure)} yang berfungsi menyalurkan beban dari struktur atas ke tanah.
	Fungsi utama:

	\begin{itemize}
		\item Menyalurkan beban vertikal
		\item Menahan gaya lateral
		\item Menjaga kestabilan bangunan
		\item Mencegah penurunan berlebih
	\end{itemize}
\end{frame}

\begin{frame}[t,mycolor=digiPH_ocean,mytitle=center,dark]
	\frametitle{Klasifikasi Pondasi}

	Pondasi dibedakan menjadi:

	\begin{itemize}
		\item Pondasi Dangkal
		\item Pondasi Dalam
	\end{itemize}

	Pemilihan ditentukan oleh:
	\begin{itemize}
		\item Kondisi tanah
		\item Kedalaman tanah keras
		\item Besar beban bangunan
		\item Jenis struktur
	\end{itemize}


\end{frame}

\section{Pondasi Dangkal}

\begin{frame}[t,mycolor=digiPH_leaf,mytitle=standard,dark]
	\frametitle{Pengertian Pondasi Dangkal}

	Digunakan apabila:

	\begin{itemize}
		\item Lapisan tanah keras dekat permukaan
		\item Beban bangunan relatif kecil–menengah
		\item Kedalaman ± 1–3 meter
	\end{itemize}

	Lebih ekonomis dan sederhana dalam pelaksanaan.

\end{frame}

\begin{frame}[c,mycolor=digiPH_leaf,mytitle=standard,dark]

	\frametitle{Perhitungan Luas Penampang Beton}

	Rumus dasar untuk menentukan dimensi kolom:

	\begin{equation}
		A_c = \frac{P_u}{\phi \, f'_c}
	\end{equation}
	Keterangan:
	\begin{itemize}
		\item $A_c$ = Luas penampang beton tekan (mm$^2$)
		\item $P_u$ = Beban aksial terfaktor (N)
		\item $\phi$ = Faktor reduksi kekuatan
		\item $f'_c$ = Kuat tekan beton (MPa)
	\end{itemize}
\end{frame}

\begin{frame}[c,mycolor=digiPH_gray,mytitle=standard,light]

	\frametitle{Perhitungan Tulangan Longitudinal}
	Tulangan longitudinal dihitung berdasarkan kebutuhan daya dukung tekan dan tarik:

	\begin{equation}
		A_s = \frac{P_u - \phi A_c f'_c}{\phi f_y}
	\end{equation}

	\vspace{0.5cm}

	Keterangan:
	\begin{itemize}
		\item $A_s$ = Luas tulangan baja (mm$^2$)
		\item $P_u$ = Beban aksial terfaktor (N)
		\item $A_c$ = Luas penampang beton (mm$^2$)
		\item $f'_c$ = Kuat tekan beton (MPa)
		\item $f_y$ = Kuat leleh baja (MPa)
		\item $\phi$ = Faktor reduksi kekuatan
	\end{itemize}\end{frame}


\begin{frame}[c,mycolor=digiPH_leaf,mytitle=standard,dark]\frametitle{Perhitungan Tulangan Sengkang}

	Jarak antar sengkang ditentukan untuk mengontrol gaya geser:

	\begin{equation}
		S = \frac{0{,}75\, d}{\sqrt{f'_c}}
	\end{equation}

	\vspace{0.5cm}

	Keterangan:
	\begin{itemize}
		\item $S$ = Jarak antar sengkang (mm)
		\item $d$ = Tinggi efektif penampang (mm)
		\item $f'_c$ = Kuat tekan beton (MPa)
	\end{itemize}
\end{frame}

\begin{frame}[c,mycolor=digiPH_ocean,mytitle=center,dark]
	\frametitle{Pemeriksaan Stabilitas Kolom (Tekuk)}

	Pastikan kolom aman terhadap tekuk dengan rasio kelangsingan:

	\begin{equation}
		k = \frac{L_e}{r}
	\end{equation}

	\vspace{0.5cm}

	Keterangan:
	\begin{itemize}
		\item $k$ = Rasio kelangsingan kolom
		\item $L_e$ = Panjang efektif kolom (mm)
		\item $r$ = Jari-jari girasi penampang (mm)
	\end{itemize}
\end{frame}
